\sbox{\InputMath}{\MathBlock{
  \mathbf{x}=(-2,-1,0,1,2)
}}
\sbox{\TargetMath}{\MathBlock{
  \begin{gathered}
    \mathbf{y}=2\mathbf{x}+3\,\mathbf{1}_5\\[1.5pt]
    \mathbf{y}=(-1,1,3,5,7)
  \end{gathered}
}}
\sbox{\OnesMath}{%
  {\fontsize{9}{11}\selectfont
   $\mathbf{1}_5=(1,1,1,1,1)$}}
\sbox{\BaselineMath}{\MathBlock{
  \begin{aligned}
    \widehat{\mathbf{y}}^{(0)}&=2\mathbf{x}\\[1.5pt]
                            &=(-4,-2,0,2,4)
  \end{aligned}
}}
\sbox{\ResidualMath}{\MathBlock{
  \begin{gathered}
    \mathbf{r}=\mathbf{y}-2\mathbf{x}\\[1.5pt]
    \mathbf{r}=(3,3,3,3,3)
  \end{gathered}
}}
\sbox{\MeanMath}{\MathBlock{
  b=\frac{1}{5}\sum_{i=1}^{5}r_i=3
}}
\sbox{\CorrectedMath}{\MathBlock{
  \begin{aligned}
    \widehat{\mathbf{y}}^{\,\mathrm{corr}}
      &=2\mathbf{x}+b\,\mathbf{1}_5\\[1.5pt]
    \widehat{\mathbf{y}}^{\,\mathrm{corr}}
      &=(-1,1,3,5,7)
  \end{aligned}
}}
\sbox{\CompareMath}{\MathBlock{
  \mathbf{y}\quad\text{vs.}\quad
  \widehat{\mathbf{y}}^{\,\mathrm{corr}}
}}
\sbox{\MSEMath}{\MathBlock{
  \begin{aligned}
    \mathrm{MSE}
      &=\frac{1}{5}\sum_{i=1}^{5}
        \left(y_i-\widehat y_i^{\,\mathrm{corr}}\right)^2\\[1.5pt]
      &=0
  \end{aligned}
}}

      minimum width=\ColA,minimum height=\LeftLowerH]
  (targets) at (\xA,-\RowSep) {\usebox{\TargetPanel}};

      minimum width=\ColB,minimum height=\UpperH]
  (baseline) at (\xB,0) {\usebox{\BaselinePanel}};

      minimum width=\ColB,minimum height=\LeftLowerH]
  (residuals) at (\xB,-\RowSep) {\usebox{\ResidualPanel}};

      minimum width=\ColC,minimum height=\MeanH]
  (mean) at (\xC,-\RowSep) {\usebox{\MeanPanel}};

\node[panel,fill=tealfill,draw=tealline,
      minimum width=\ColD,minimum height=\UpperH]

      minimum width=\ColD,minimum height=\EvalH]
  (evaluation) at (\xD,-\RowSep) {\usebox{\EvaluationPanel}};

% 2. Baseline values used to calculate observed residuals.

% 3. Observed targets used to calculate residuals.

% 4. Residual averaging.
\draw[adjustment]
  (residuals.east)
  -- node[edge label,midway,above=1.5mm]
       {\usebox{\AverageText}}
  (mean.west);

% 5. Baseline contribution to the corrected prediction.
\draw[supplied]
  ($(baseline.east)+(0,3.5mm)$)
  -- node[edge label,midway,above=1.5mm]
       {\MathBlock{2\mathbf{x}}}
  ($(corrected.west)+(0,3.5mm)$);

% 6. Scalar correction, entering through a separate lower left port.
\draw[adjustment,rounded corners=1.6mm]
  (mean.north)
  -- (\xC,-5.5mm)
  -- node[edge label,midway,above=1.5mm]
       {\MathBlock{b=3}}
  ($(corrected.west)+(0,-5.5mm)$);

% 7. Corrected values passed to evaluation.
\draw[supplied] (corrected.south) -- (evaluation.north);

% 8. A dedicated channel carries the same observed targets to evaluation.

  (targets.south)
  -- (\xA,\ChannelY)
  -- node[edge label,midway,below=1.5mm]
       {observed targets: same five points}
  (\xD,\ChannelY)
  -- (evaluation.south);
