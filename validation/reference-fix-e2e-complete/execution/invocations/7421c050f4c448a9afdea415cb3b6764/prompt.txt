  geom_vline(xintercept = df$b, colour = "#E3E3E3",
             linewidth = 0.3, linetype = "dashed") +
  geom_point(shape = 21, size = 4, fill = "#1F4E79",
             colour = "#FFFFFF", stroke = 0.6) +
  geom_text(aes(label = paste0("MSE = ", format(mse, drop0trailing = TRUE))),

  labs(
    title   = "Measured Mean Squared Error at Three Intercept Corrections",
    x       = "Added intercept correction b",
    y       = "Mean squared error (MSE)",
    caption = "Each point is one deterministic evaluation on the same five observations used to determine b. No uncertainty is shown."
  ) +
  theme_minimal(base_family = "Helvetica") +
  theme(
